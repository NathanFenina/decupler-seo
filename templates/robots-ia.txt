#  Configuration recommandée pour la plupart des sites : rester
#  visible dans les moteurs IA.
#
#  Distinction importante : l'entraînement et la recherche sont deux
#  choses différentes. Vous pouvez refuser d'alimenter l'entraînement
#  tout en restant citable dans les réponses.
# ─────────────────────────────────────────────────────────────────

# Navigation à la demande d'un utilisateur — il a explicitement
# demandé votre page ; la bloquer n'a aucun sens

# Perplexity — le moteur IA qui cite le plus systématiquement

Allow: /

# Gemini et AI Overviews.
